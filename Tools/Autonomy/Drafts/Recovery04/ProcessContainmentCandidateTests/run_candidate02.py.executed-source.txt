class Lane:
    def __init__(self, folder):
        self.started = time.monotonic()
        self.safe_end = self.started + 90
        self.deadline = datetime.now(timezone.utc) + timedelta(seconds=180)
        self.events, self.logs, self.env = [], [], dict(os.environ, PYTHONDONTWRITEBYTECODE='1', PYTHONOPTIMIZE='0')
        self.run = folder / 'Run'
        self.run.mkdir(parents=True)

    def remaining(self):
        remaining = min(self.safe_end - time.monotonic(), (self.deadline - datetime.now(timezone.utc)).total_seconds() - 60)
        if remaining <= 0:
            raise TimeoutError('Isolated test budget exhausted')
        return remaining


def current(pid):
    p = Path('/proc') / str(pid)
    raw = (p / 'stat').read_text().rsplit(') ', 1)[1].split()
    return {'state': raw[0], 'startTicks': int(raw[19]), 'uid': p.stat().st_uid}


def live(row):
    try:
        actual = current(row['pid'])
        return actual['state'] not in ('Z', 'X') and actual['startTicks'] == row['startTicks'] and actual['uid'] == row['uid']
    except (FileNotFoundError, ProcessLookupError):
        return False


def cleanup_fixture_rows(folder, token):
    cleaned = []
    for p in folder.glob('*.json'):
        row = json.loads(p.read_text())
        if row.get('token') != token or not live(row):
            continue
        cmd = (Path('/proc') / str(row['pid']) / 'cmdline').read_bytes()
        assert str(fixture).encode() in cmd and token.encode() in cmd
        fd = os.pidfd_open(row['pid'])
        try:
            assert live(row)
            signal.pidfd_send_signal(fd, signal.SIGKILL)
            cleaned.append(row)
        finally:
            os.close(fd)
    return cleaned
